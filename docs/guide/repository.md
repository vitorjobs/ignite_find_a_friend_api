```bash
find-a-friend-api/
├── src/ # Código fonte
├── docker/ # Configurações Docker
├── docs/ # Documentação
├── build/ # Arquivos de build
├── coverage/ # Relatórios de cobertura de testes

├── .env.example # Exemplo de variáveis de ambiente
```
